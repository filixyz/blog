{{- /* Banner image from params.header: a file in assets/ (any size, 4:1 works best, 1600x400 ideal). Hugo makes 600/1200/1600px WebP and JPEG copies whose file names contain a hash of the picture, so a new banner automatically gets new URLs while the old ones stay cached for a year (see static/_headers). Called once for the sidebar (desktop) and once under the top bar (mobile); CSS shows the right one. Both ask for the same file, so it downloads once. */ -}} {{- with site.Params.header }}{{ with resources.Get . }} {{- $base := resources.Copy (printf "banner/%s" .Name) . -}} {{- $ws := slice }}{{ $js := slice }} {{- range slice 600 1200 1600 }} {{- $w := . }} {{- if le $w $base.Width }} {{- $ws = $ws | append (printf "%s %dw" ($base.Fill (printf "%dx%d webp q80 Center" $w (div $w 4))).RelPermalink $w) }} {{- $js = $js | append (printf "%s %dw" ($base.Fill (printf "%dx%d jpg q82 Center" $w (div $w 4))).RelPermalink $w) }} {{- end }} {{- end }} {{- $fallback := $base.Fill "1200x300 jpg q82 Center" }} {{- $sizes := "(max-width: 760px) 100vw, 25vw" -}} {{- end }}{{ end -}}
